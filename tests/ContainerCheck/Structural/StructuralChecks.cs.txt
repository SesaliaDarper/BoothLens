sealed class FakeRemote : HttpMessageHandler
{
    public List<string> Searches { get; } = [];
    public int ItemRequests { get; private set; }
    protected override Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken ct)
    {
        var uri = request.RequestUri!; var path = Uri.UnescapeDataString(uri.AbsolutePath);
        if (uri.Host == "booth.pximg.net") return Task.FromResult(new HttpResponseMessage(HttpStatusCode.OK) { Content = new ByteArrayContent([255, 216, 255, 217]) });
        if (path.Contains("/search/"))
        {
            Searches.Add(path);
            return Task.FromResult(new HttpResponseMessage(HttpStatusCode.OK) { Content = new StringContent("<li class='item-card' data-product-id='2345678' data-product-name='Candidate' data-product-brand='test' data-product-price='100'></li>") });
        }
        ItemRequests++;
        var value = new { id = 1234567, name = "Verified remote", url = "https://test.booth.pm/items/1234567", images = new[] { new { original = "https://booth.pximg.net/remote.jpg" } }, shop = new { name = "Test", subdomain = "test" } };
        return Task.FromResult(new HttpResponseMessage(HttpStatusCode.OK) { Content = new StringContent(JsonSerializer.Serialize(value), Encoding.UTF8, "application/json") });
    }
}
